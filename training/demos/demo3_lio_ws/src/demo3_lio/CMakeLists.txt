cmake_minimum_required(VERSION 3.8)
project(demo3_lio)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_executable(lio_student_node src/lio_student_node.cpp)
add_executable(lio_student_controller src/lio_student_controller.cpp)
target_include_directories(lio_student_node PRIVATE include)
target_include_directories(lio_student_controller PRIVATE include)
ament_target_dependencies(lio_student_node rclcpp builtin_interfaces sensor_msgs nav_msgs geometry_msgs tf2 tf2_ros)
ament_target_dependencies(lio_student_controller rclcpp sensor_msgs nav_msgs geometry_msgs tf2 tf2_ros tf2_geometry_msgs)
install(TARGETS lio_student_node lio_student_controller DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(DIRECTORY rviz DESTINATION share/${PROJECT_NAME})
ament_package()
